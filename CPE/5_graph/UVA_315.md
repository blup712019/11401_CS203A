## UVa 315 - Network
- 類型：Graph / DFS / Articulation Point
- 解法思路：
  1. 題目要找「割點」：移除該點以及相關邊會讓圖的連通分量數增加的點。
  2. 使用 Tarjan DFS：
     - disc[u]：DFS 走到 u 的時間戳
     - low[u]：u 或 u 的子樹，透過樹邊+回邊能到達的最小 disc
  3. 割點判定：
     - 若 u 不是 DFS root，且存在子節點 v 使得 low[v] >= disc[u]，則 u 是割點
     - 若 u 是 DFS root，且 root 的 DFS 子樹數 >= 2，則 root 是割點

### 參考實作

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    while (cin >> n && n) {
        cin.ignore(numeric_limits<streamsize>::max(), '\n');

        vector<vector<int>> adj(n + 1);
        string line;

        while (getline(cin, line)) {
            if (line == "0") break;
            stringstream ss(line);
            int u;
            ss >> u;
            int v;
            while (ss >> v) {
                adj[u].push_back(v);
                adj[v].push_back(u);
            }
        }

        vector<int> disc(n + 1, 0), low(n + 1, 0), parent(n + 1, -1);
        vector<bool> isArt(n + 1, false);
        int timer = 0;

        function<void(int)> dfs = [&](int u) {
            disc[u] = low[u] = ++timer;
            int child = 0;

            for (int v : adj[u]) {
                if (!disc[v]) {
                    parent[v] = u;
                    child++;
                    dfs(v);
                    low[u] = min(low[u], low[v]);

                    if (parent[u] == -1 && child >= 2) isArt[u] = true;
                    if (parent[u] != -1 && low[v] >= disc[u]) isArt[u] = true;
                } else if (v != parent[u]) {
                    low[u] = min(low[u], disc[v]);
                }
            }
        };

        for (int i = 1; i <= n; i++) {
            if (!disc[i]) dfs(i);
        }

        int ans = 0;
        for (int i = 1; i <= n; i++) if (isArt[i]) ans++;
        cout << ans << "\n";
    }
    return 0;
}
