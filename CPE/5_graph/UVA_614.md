## UVa 614 - Mapping the Route
- 類型：Graph / DFS / Maze Simulation
- 解法思路：
  1. 迷宮每格只給「東牆(1)」與「南牆(2)」，3 代表兩者都有；外圍牆題目保證存在，不會在輸入給。 :contentReference[oaicite:1]{index=1}
  2. 依題目指定的 DFS 走法：在每個格子依序嘗試往 西、北、東、南 走。能走的條件是「沒有牆」且「下一格沒走過」。走到底沒路就回退。 :contentReference[oaicite:2]{index=2}
  3. 走訪過但最後不在起點到終點路徑上的格子要輸出 `???`；路徑上的格子輸出從 1 開始的序號；完全沒走到的格子輸出空白。 :contentReference[oaicite:3]{index=3}
  4. 輸出時照題目格式畫出邊框與牆（`|` 與 `---`）。

- 複雜度：O(h*w)（每格最多被走訪一次）

### 參考實作

```cpp
#include <iostream>
#include <vector>
#include <fstream>
#include <iomanip>
using namespace std;
int h, w, sy, sx, py, px,maze=0;
int map[15][15];
int found[15][15];
bool dfs(int y,int x,int nums) {
	
	if (found[y][x] == 0) {
		nums++;
		found[y][x] = -1;
		if (y == py && x == px) {
			found[y][x] = nums;
			return 1;
		}
		if (map[y][x - 1] != 1 && map[y][x - 1] != 3 && map[y][x - 1] != -1) {
			if (dfs(y, x - 1, nums)) {
				found[y][x] = nums;
				return 1;
			}
		}
		if (map[y - 1][x] != 2 && map[y-1][x] != 3 && map[y-1][x] != -1) {
			if (dfs(y-1, x, nums)) {
				found[y][x] = nums;
				return 1;
			}
		}
		if (map[y][x] != 1 && map[y][x] != 3 && map[y][x + 1] != -1) {
			if (dfs(y, x + 1, nums)) {
				found[y][x] = nums;
				return 1;
			}
		}
		if (map[y][x] != 2 && map[y][x] != 3 && map[y + 1][x] != -1) {
			if (dfs(y + 1, x, nums)) {
				found[y][x] = nums;
				return 1;
			}
		}
	}
	return 0;
}
int main() {
	std::cin >> h >> w >> sy >> sx >> py >> px;
	while (h != 0 && w != 0 && sy != 0 && sx != 0 && py != 0 && px != 0) {
		maze++;
		for (int i = 0; i < 15; i++) {
			for (int j = 0; j < 15; j++) {
				map[j][i] = -1;
				found[j][i] = 0;
			}
		}
		for (int i = 1; i <= h; i++) {
			for (int j = 1; j <= w; j++) {
				std::cin >> map[i][j];
			}
		}
		int nums = 0;
		dfs(sy, sx, nums);
		cout << "Maze " << maze << std::endl << std::endl;
		for (int i = 0; i <= w * 4; i++) {
			if (i % 4 == 0) {
				cout << "+";
			}
			else {
				cout << "-";
			}
		}
		cout << std::endl;
		for (int i = 1; i <= h; i++) {
			cout << "|";
			for (int j = 1; j <=  w; j++) {
				if (found[i][j] == -1) {
					cout << "???";
				}
				else if (found[i][j] == 0) {
					cout << "   ";
				}
				else {
					cout << setw(3) << found[i][j];
				}
				if (j != w) {
					if (map[i][j] == 1 || map[i][j] == 3) {
						cout << "|";
					}
					else {
						cout << " ";
					}
				}
				
			}
			cout << "|";
			cout << std::endl;
			if (i != h) {
				cout << "+";
				for (int j = 1; j <= w; j++) {
					if (map[i][j] == 2 || map[i][j] == 3) {
						cout << "---+";
					}
					else {
						cout << "   +";
					}
				}
				cout << std::endl;
			}
		}
		for (int i = 0; i <= w * 4; i++) {
			if (i % 4 == 0) {
				cout << "+";
			}
			else {
				cout << "-";
			}
		}
		cout << std::endl << std::endl << std::endl;
		std::cin >> h >> w >> sy >> sx >> py >> px;
	}
}
