## UVa 871 - Counting Cells in a Blob
- 類型：Graph / DFS / Flood Fill（8-direction）
- 解法思路：
  1. 輸入是一個由 '0'/'1' 組成的字元矩陣（用多行字串表示），測資之間以空行分隔。
  2. 8 方向（上下左右 + 四個對角）相鄰的 '1' 視為同一個 blob。
  3. 走訪整張圖：遇到尚未走過的 '1' 就做 DFS，把同一個 blob 的大小算出來並更新最大值。
  4. 輸出每個測資最大 blob 的大小，測資之間空一行。
- 複雜度：O(R*C)

### 參考實作

```cpp
#include <iostream>
#include <vector>
#include <string>
using namespace std;

int dr[8] = {-1,-1,-1,0,0,1,1,1};
int dc[8] = {-1,0,1,-1,1,-1,0,1};

vector<string> g;
vector< vector<int> > vis;
int R, C;

bool inside(int x, int y) {
    return x >= 0 && x < R && y >= 0 && y < C;
}

int dfs(int x, int y) {
    vis[x][y] = 1;
    int cnt = 1;
    for (int k = 0; k < 8; k++) {
        int nx = x + dr[k], ny = y + dc[k];
        if (inside(nx, ny) && !vis[nx][ny] && g[nx][ny] == '1') {
            cnt += dfs(nx, ny);
        }
    }
    return cnt;
}

int main() {
    int T;
    cin >> T;
    string line;

    getline(cin, line); // 吃掉 T 那行剩下的換行
    getline(cin, line); // 吃掉空白行（題目格式通常有）

    while (T--) {
        g.clear();

        while (getline(cin, line)) {
            if (line.size() == 0) break;
            g.push_back(line);
        }

        R = (int)g.size();
        C = R ? (int)g[0].size() : 0;

        vis.assign(R, vector<int>(C, 0));

        int ans = 0;
        for (int i = 0; i < R; i++) {
            for (int j = 0; j < C; j++) {
                if (!vis[i][j] && g[i][j] == '1') {
                    int size = dfs(i, j);
                    if (size > ans) ans = size;
                }
            }
        }

        cout << ans << "\n";
        if (T) cout << "\n";
    }
    return 0;
}
