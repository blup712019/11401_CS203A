## UVa 280 - Vertex
- 類型：Graph / DFS / Reachability
- 解法思路：
  1. 題目給一張有向圖，節點編號為 1~N。
  2. 對於每一個查詢起點 s，要找出「從 s 出發無法到達的節點」。
  3. 對每個查詢：
     - 初始化 visited 陣列
     - 從 s 做 DFS（或 BFS），標記所有可到達的節點
     - 將所有 visited 為 false 的節點輸出
  4. 特別注意：起點 s 本身若無法回到自己，也要算在不可達內。
- 複雜度：
  - 建圖：O(E)
  - 每次查詢 DFS：O(N + E)

### 參考實作

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {
    int n;
    while (cin >> n && n) {
        vector<vector<int>> adj(n + 1);

        while (true) {
            int u;
            cin >> u;
            if (u == 0) break;
            while (true) {
                int v;
                cin >> v;
                if (v == 0) break;
                adj[u].push_back(v);
            }
        }

        int q;
        cin >> q;
        while (q--) {
            int s;
            cin >> s;

            vector<bool> vis(n + 1, false);
            stack<int> st;
            st.push(s);

            while (!st.empty()) {
                int u = st.top();
                st.pop();
                for (int v : adj[u]) {
                    if (!vis[v]) {
                        vis[v] = true;
                        st.push(v);
                    }
                }
            }

            vector<int> unreachable;
            for (int i = 1; i <= n; i++) {
                if (!vis[i]) unreachable.push_back(i);
            }

            cout << unreachable.size();
            for (int x : unreachable) cout << " " << x;
            cout << "\n";
        }
    }
    return 0;
}
