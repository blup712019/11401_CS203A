## UVa 10354 - Avoiding Your Boss

**類型**：Graph / Shortest Path / Dijkstra

**解法思路**：

- 圖為無向帶權
- 老闆從 BH 到 OF 會走「最短路徑」（可能有多條），必須避開所有可能出現在任一條最短路徑上的點
- 先跑兩次 Dijkstra：
  - distBH：BH 到各點最短距離
  - distOF：OF 到各點最短距離
- 令 bossDist = distBH[OF]
  - 若 distBH[i] + distOF[i] == bossDist，則 i 在某條最短路徑上，標記為禁用
  - 額外把 BH、OF 也視為禁用
- 再跑一次 Dijkstra 從 YH 到 M，途中不可走到禁用點
- 若不可達輸出 `MISSION IMPOSSIBLE.`

**複雜度**：

- 時間：O((P+R) log P)
- 空間：O(P+R)

### 參考實作

```cpp
#include <iostream>
#include <vector>
#include <queue>
#include <utility>
#include <limits>
using namespace std;

using ll = long long;
static const ll INF = (1LL << 60);

static vector<ll> dijkstra(int n, const vector<vector<pair<int,int>>>& g, int src, const vector<char>& banned) {
    vector<ll> dist(n + 1, INF);
    if (src < 1 || src > n) return dist;
    if (banned[src]) return dist;

    priority_queue<pair<ll,int>, vector<pair<ll,int>>, greater<pair<ll,int>>> pq;
    dist[src] = 0;
    pq.push({0, src});

    while (!pq.empty()) {
        auto cur = pq.top();
        pq.pop();
        ll d = cur.first;
        int u = cur.second;
        if (d != dist[u]) continue;

        for (auto e : g[u]) {
            int v = e.first;
            int w = e.second;
            if (banned[v]) continue;
            ll nd = d + (ll)w;
            if (nd < dist[v]) {
                dist[v] = nd;
                pq.push({nd, v});
            }
        }
    }
    return dist;
}

int main() {
    int p, r, bh, of, yh, m;
    while (cin >> p >> r >> bh >> of >> yh >> m) {
        vector<vector<pair<int,int>>> g(p + 1);
        for (int i = 0; i < r; i++) {
            int a, b, d;
            cin >> a >> b >> d;
            g[a].push_back({b, d});
            g[b].push_back({a, d});
        }

        vector<char> none(p + 1, 0);
        vector<ll> distBH = dijkstra(p, g, bh, none);
        vector<ll> distOF = dijkstra(p, g, of, none);

        vector<char> banned(p + 1, 0);
        banned[bh] = 1;
        banned[of] = 1;

        ll bossDist = distBH[of];
        if (bossDist < INF / 2) {
            for (int i = 1; i <= p; i++) {
                if (distBH[i] >= INF / 2 || distOF[i] >= INF / 2) continue;
                if (distBH[i] + distOF[i] == bossDist) banned[i] = 1;
            }
        }

        vector<ll> distYH = dijkstra(p, g, yh, banned);
        if (yh < 1 || yh > p || m < 1 || m > p || banned[yh] || banned[m] || distYH[m] >= INF / 2) {
            cout << "MISSION IMPOSSIBLE.\n";
        } else {
            cout << distYH[m] << "\n";
        }
    }
    return 0;
}
```
