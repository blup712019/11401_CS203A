## UVa 429 - Word Transformation
- 類型：Graph / BFS / Shortest Path（無權圖最短路）
- 解法思路：
  1. 先讀字典（每行一個單字，遇到 `*` 結束）。
  2. 兩個單字若「長度相同且只有一個字母不同」，就可視為圖上的一條無向邊。
  3. 對每個查詢 `(s, t)`，在圖上做 BFS，求從 `s` 到 `t` 的最短步數（每次轉換一步）。
  4. 輸出 `s t distance`。
- 複雜度：
  - 建圖（你這版做法）：O(W^2 * L)（W=單字數，L=單字長度）
  - 每次查詢 BFS：O(W + E)

### 參考實作

```cpp
#include <iostream>
#include <string>
#include <vector>
#include <map>
#include <queue>
#include <sstream>

using namespace std;

map<string, vector<string>> g;

static bool one_diff(const string& a, const string& b) {
    if (a.size() != b.size()) return false;
    int diff = 0;
    for (size_t i = 0; i < a.size(); i++) {
        if (a[i] != b[i]) {
            diff++;
            if (diff > 1) return false;
        }
    }
    return diff == 1;
}

static int bfs_dist(const string& s, const string& t) {
    map<string, int> dist;
    for (auto& kv : g) dist[kv.first] = -1;

    queue<string> q;
    dist[s] = 0;
    q.push(s);

    while (!q.empty()) {
        string u = q.front();
        q.pop();
        if (u == t) break;

        for (const auto& v : g[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }
    return dist[t];
}

int main() {
    int tc;
    cin >> tc;

    for (int cs = 0; cs < tc; cs++) {
        g.clear();

        string word;
        while (cin >> word && word != "*") {
            g[word];
            for (auto& kv : g) {
                const string& w2 = kv.first;
                if (word == w2) continue;
                if (one_diff(word, w2)) {
                    g[word].push_back(w2);
                    g[w2].push_back(word);
                }
            }
        }

        cin.ignore();

        string line;
        while (getline(cin, line)) {
            if (line.empty()) break;
            string s, t;
            stringstream ss(line);
            ss >> s >> t;
            cout << s << " " << t << " " << bfs_dist(s, t) << "\n";
        }

        if (cs != tc - 1) cout << "\n";
    }
    return 0;
}
