## UVa 10229 - Modular Fibonacci

**類型**：Math / Matrix Exponentiation / Fibonacci Mod

**解法思路**：

- 要求輸出 F(n) mod 2^m
- 令 mod = 2^m（用位移算出）
- 利用 Fibonacci 的線性遞迴可用快速冪加速到 O(log n)
- 每次運算都取 mod，避免數值爆掉

**複雜度**：

- 時間：O(log n)
- 空間：O(1)

### 參考實作

```cpp
#include <iostream>
#include <vector>
#include <algorithm>
#define int long long
using namespace std;
int moder;
vector<int> polymut(vector<int> right, vector<int> left) {
    vector<int> temp(right.size()+left.size());
    for (int i = 0; i < right.size(); i++) {
        for (int j = 0; j < left.size(); j++) {
            temp[i + j] = (temp[i + j] + right[i] * left[j]) % moder;
        }
    }
    for (int i = temp.size() - 1; i >= 2; i--) {
        temp[i - 1] =(temp[i-1] + temp[i]) % moder;
        temp[i - 2] = (temp[i - 2] + temp[i] )% moder;
        temp.pop_back();
    }
    return temp;
}
signed main() {
    int n, m;
    while (cin >> n >> m) {
        moder = 1;
        moder = (moder << m);
        vector<int> res = { 1, 0 };       
        vector<int> base = { 0, 1 };
        int e = n;
        while (e > 0) {
            if (e % 2 == 1) res = polymut(res, base);
            base = polymut(base, base);            
            e /= 2;
        }
        cout << (res[1] % moder) << "\n";
    }

}
```
