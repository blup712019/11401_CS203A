## UVa 541 - Error Correction
- 類型：Matrix / Parity / Simulation
- 解法思路：
  1. 讀入 n×n 0/1 矩陣。
  2. 分別計算每一列與每一行的 1 的個數是否為奇數（parity 錯誤）。
  3. 若沒有任何奇數列/行 → `OK`
  4. 若剛好有一個奇數列 + 一個奇數行 → 那個交點翻轉即可，輸出 `Change bit (row,col)`（1-based）
  5. 其他情況 → `Corrupt`
- 複雜度：O(n^2)

### 參考實作

```cpp
#include <iostream>
#include <vector>
using namespace std;

int main() {
    int n;
    while (cin >> n && n) {
        vector<vector<int>> a(n, vector<int>(n));
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) cin >> a[i][j];
        }

        int oddRow = -1, oddCol = -1;
        int oddRowCnt = 0, oddColCnt = 0;

        for (int i = 0; i < n; i++) {
            int s = 0;
            for (int j = 0; j < n; j++) s += a[i][j];
            if (s % 2) {
                oddRow = i;
                oddRowCnt++;
            }
        }

        for (int j = 0; j < n; j++) {
            int s = 0;
            for (int i = 0; i < n; i++) s += a[i][j];
            if (s % 2) {
                oddCol = j;
                oddColCnt++;
            }
        }

        if (oddRowCnt == 0 && oddColCnt == 0) {
            cout << "OK\n";
        } else if (oddRowCnt == 1 && oddColCnt == 1) {
            cout << "Change bit (" << (oddRow + 1) << "," << (oddCol + 1) << ")\n";
        } else {
            cout << "Corrupt\n";
        }
    }
    return 0;
}
