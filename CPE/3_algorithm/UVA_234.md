## UVa 234 - Switching Channels
- 類型：Brute Force / Permutation / Simulation
- 解法思路：
  1. 給定 N 個節目長度，還有 M 個對齊點（alignment points），每個對齊點有等級 level (1~5) 與時間 ap。
  2. 選定一個播放順序後，會形成切換時間點 cuts：0、播完第1個後、播完第2個後…直到總長度。
  3. 對每個對齊點 (level, ap)，計算它到最近切換時間點的距離：min |ap - cuts[i]|，把距離累加到 error[level]。
  4. 目標是最小化 error[1]，若相同再比 error[2]，一路比到 error[5]（字典序比較）。
  5. 做法：把節目長度排序後，用 next_permutation 枚舉所有不同排列，對每個排列計算 error 向量，取最優。
- 複雜度：
  - 枚舉：O(N!)
  - 每個排列計算：O((N+M) log N)（用二分找最近切換點）
  - 總體：O(N! * (N+M) log N)

### 參考實作

```cpp
#include <bits/stdc++.h>
using namespace std;

int main() {

    int n;
    int tc = 1;
    while (cin >> n && n) {
        vector<int> a(n);
        for (int i = 0; i < n; i++) cin >> a[i];

        int m;
        cin >> m;
        vector<pair<int,int>> pts(m);
        for (int i = 0; i < m; i++) {
            int level, ap;
            cin >> level >> ap;
            pts[i] = {ap, level};
        }

        sort(a.begin(), a.end());

        vector<long long> bestErr(6, (1LL<<60));
        vector<int> bestOrder = a;

        auto calcErr = [&](const vector<int>& order) -> vector<long long> {
            vector<long long> cuts;
            cuts.reserve(order.size() + 1);
            long long s = 0;
            cuts.push_back(0);
            for (int x : order) {
                s += x;
                cuts.push_back(s);
            }

            vector<long long> err(6, 0);
            for (auto [ap, level] : pts) {
                long long p = ap;
                auto it = lower_bound(cuts.begin(), cuts.end(), p);
                long long d = (1LL<<60);
                if (it != cuts.end()) d = min(d, llabs(*it - p));
                if (it != cuts.begin()) {
                    auto it2 = prev(it);
                    d = min(d, llabs(*it2 - p));
                }
                err[level] += d;
            }
            return err;
        };

        auto better = [&](const vector<long long>& x, const vector<long long>& y) -> bool {
            for (int lv = 1; lv <= 5; lv++) {
                if (x[lv] != y[lv]) return x[lv] < y[lv];
            }
            return false;
        };

        do {
            auto e = calcErr(a);
            if (better(e, bestErr)) {
                bestErr = e;
                bestOrder = a;
            }
        } while (next_permutation(a.begin(), a.end()));

        cout << "Data set " << tc << "\n";
        cout << "  Order: ";
        for (int x : bestOrder) cout << x << " ";
        cout << "\n";
        long long total = 0;
        for (int lv = 1; lv <= 5; lv++) total += bestErr[lv];
        cout << "  Error: " << total << "\n";
        tc++;
    }
    return 0;
}
