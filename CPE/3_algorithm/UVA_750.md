## UVa 750 - 8 Queens Chess Problem
- 類型：Backtracking / Constraint Checking
- 解法思路：
  1. 題目給一個固定皇后位置 (r, c)，要列出所有 8 皇后解且該位置必須放皇后。
  2. 用回溯以「欄」為單位放皇后：每一欄放一個皇后。
  3. 用三個集合/陣列檢查衝突：
     - row[r]：該列是否已放
     - diag1[r - c + 7]：主對角線
     - diag2[r + c]：副對角線
  4. 走到第 8 欄就得到一組解，輸出時印出各欄的列號（1..8）。
  5. 輸出格式需包含表頭，解要依照題目慣例排序（回溯自然順序即可），不同測資之間空一行。

### 參考實作

```cpp
#include <iostream>
#include <vector>
using namespace std;

vector<vector<int>> sols;
int fixedR, fixedC;
bool usedRow[9], usedD1[20], usedD2[20];
int pos[9];

void dfs(int c) {
    if (c == 9) {
        if (pos[fixedC] == fixedR) sols.push_back(vector<int>(pos + 1, pos + 9));
        return;
    }
    if (c == fixedC) {
        int r = fixedR;
        if (!usedRow[r] && !usedD1[r - c + 8] && !usedD2[r + c]) {
            usedRow[r] = usedD1[r - c + 8] = usedD2[r + c] = true;
            pos[c] = r;
            dfs(c + 1);
            usedRow[r] = usedD1[r - c + 8] = usedD2[r + c] = false;
        }
        return;
    }
    for (int r = 1; r <= 8; r++) {
        if (!usedRow[r] && !usedD1[r - c + 8] && !usedD2[r + c]) {
            usedRow[r] = usedD1[r - c + 8] = usedD2[r + c] = true;
            pos[c] = r;
            dfs(c + 1);
            usedRow[r] = usedD1[r - c + 8] = usedD2[r + c] = false;
        }
    }
}

int main() {
    int T;
    cin >> T;
    for (int tc = 0; tc < T; tc++) {
        cin >> fixedR >> fixedC;

        sols.clear();
        for (int i = 1; i <= 8; i++) usedRow[i] = false;
        for (int i = 0; i < 20; i++) usedD1[i] = usedD2[i] = false;

        dfs(1);

        cout << "SOLN       COLUMN\n";
        cout << " #      1 2 3 4 5 6 7 8\n\n";
        for (int i = 0; i < (int)sols.size(); i++) {
            cout << " " << (i + 1) << "     ";
            for (int c = 1; c <= 8; c++) {
                cout << " " << sols[i][c - 1];
            }
            cout << "\n";
        }
        if (tc != T - 1) cout << "\n";
    }
    return 0;
}
